const { expect } = require('@playwright/test');

class HeaderComponent {
    constructor(page) {
        this.page = page;

        // Complete site Header
        this.header = page.locator('header#header');

        this.mainNavigation = this.header.locator(
            '#block-main-navigation'
        );

        // Home logo
        this.homeLogo = this.header.locator(
            'a.navbar-brand[rel="home"]'
        );

        this.homeLogoImage = this.homeLogo.locator(
            'img'
        );

        // Main navigation links
        this.productsLink = this.mainNavigation.locator(
            'a[href="/products"]'
        );

        this.whoWeAreDropdown = this.mainNavigation.locator(
            'li.menu-item--expanded.dropdown'
        );

        this.whoWeAreMenu = this.whoWeAreDropdown.locator(
            'a[data-bs-toggle="dropdown"]'
        ).first();

        this.whoWeAreDropdownMenu =
            this.whoWeAreDropdown.locator(
                'ul.dropdown-menu'
            );

        // Stable Who We Are submenu links
        this.aboutUsLink = this.whoWeAreDropdown.locator(
            'a[href="/about-us"]'
        );

        this.foodCategoryLink =
            this.whoWeAreDropdown.locator(
                'a[href="/food-1-category"]'
            );

        this.locationLink =
            this.whoWeAreDropdown.locator(
                'a[href="/location"]'
            );

        // Catalog and Contact Us
        this.catalogLink = this.mainNavigation.locator(
            'a[href="/catalog"]'
        );

        this.contactUsLink = this.mainNavigation.locator(
            'a[href="/contact-us"]'
        );

        // Public B2B Login link
        this.b2bLoginLink = this.header.locator(
            'a[href="/login-modal"]'
        ).first();

        // Search controls
        this.searchButton = this.header.locator(
            'button.search__toggler'
        );

        this.searchInput = this.header.locator(
            'form.simple-search-form input[name="search"]'
        );

        // Mobile navigation
        this.mobileMenuButton = this.header.locator(
            'button.navbar-toggler'
        );

        this.collapsingNavigation = this.header.locator(
            '#CollapsingNavbar'
        );

        // OneTrust controls
        this.oneTrustBanner = page.locator(
            '#onetrust-banner-sdk'
        );

        this.oneTrustPreferenceCenter = page.locator(
            '#onetrust-pc-sdk'
        );

        this.oneTrustDarkOverlay = page.locator(
            '.onetrust-pc-dark-filter'
        );

        this.acceptAllCookiesButton = page.locator(
            '#onetrust-accept-btn-handler'
        );

        this.rejectAllCookiesButton = page.locator(
            '#onetrust-reject-all-handler'
        );

        this.acceptRecommendedButton = page.locator(
            '#accept-recommended-btn-handler'
        );

        this.savePreferencesButton = page.locator(
            '.save-preference-btn-handler'
        );

        this.closePreferencesButton = page.locator(
            '#close-pc-btn-handler'
        );
    }

    async isVisible(locator, timeout = 1500) {
        return locator.isVisible({
            timeout
        }).catch(() => false);
    }

    async clickIfVisible(locator) {
        const visible = await this.isVisible(
            locator
        );

        if (!visible) {
            return false;
        }

        await locator.click({
            force: true,
            timeout: 5000
        }).catch(() => {});

        return true;
    }

    async ensureOneTrustIsClosed() {
        const preferenceCenterVisible =
            await this.isVisible(
                this.oneTrustPreferenceCenter
            );

        const overlayVisible =
            await this.isVisible(
                this.oneTrustDarkOverlay
            );

        if (
            preferenceCenterVisible ||
            overlayVisible
        ) {
            const acceptedRecommended =
                await this.clickIfVisible(
                    this.acceptRecommendedButton
                );

            if (!acceptedRecommended) {
                const preferencesSaved =
                    await this.clickIfVisible(
                        this.savePreferencesButton
                    );

                if (!preferencesSaved) {
                    await this.clickIfVisible(
                        this.closePreferencesButton
                    );
                }
            }
        }

        const bannerVisible =
            await this.isVisible(
                this.oneTrustBanner
            );

        if (bannerVisible) {
            const cookiesAccepted =
                await this.clickIfVisible(
                    this.acceptAllCookiesButton
                );

            if (!cookiesAccepted) {
                await this.clickIfVisible(
                    this.rejectAllCookiesButton
                );
            }
        }

        await this.oneTrustPreferenceCenter.waitFor({
            state: 'hidden',
            timeout: 5000
        }).catch(() => {});

        await this.oneTrustDarkOverlay.waitFor({
            state: 'hidden',
            timeout: 5000
        }).catch(() => {});

        const overlayStillVisible =
            await this.isVisible(
                this.oneTrustDarkOverlay
            );

        if (overlayStillVisible) {
            await this.page.evaluate(() => {
                document
                    .querySelectorAll(
                        '.onetrust-pc-dark-filter'
                    )
                    .forEach(overlay => {
                        overlay.style.display = 'none';
                        overlay.style.visibility = 'hidden';
                        overlay.style.pointerEvents = 'none';
                    });
            });

            await this.oneTrustDarkOverlay.waitFor({
                state: 'hidden',
                timeout: 5000
            }).catch(() => {});
        }
    }

    async verifyHeaderDisplayed() {
        await expect(
            this.header,
            'Site Header should be displayed'
        ).toBeVisible();
    }

    async verifyLogoDisplayed() {
        await expect(

const { expect } = require('@playwright/test');

const {
    HeaderComponent
} = require('./components/HeaderComponent');

const {
    FooterComponent
} = require('./components/FooterComponent');

const {
    CookieConsentComponent
} = require('./components/CookieConsentComponent');

class HomePage {
    constructor(page) {
        this.page = page;

        // Reusable Header and Footer components
        this.header = new HeaderComponent(page);
        this.footer = new FooterComponent(page);
        this.cookieConsent = new CookieConsentComponent(page);

        // Home page breadcrumb
        this.breadcrumb = page.locator(
            'nav[aria-label="breadcrumb"]'
        );

        // Hero banner
        this.heroSection = page.locator('#block-hero');

        this.heroHeading = this.heroSection.getByRole(
            'heading',
            {
                level: 1
            }
        );

        this.heroSubtitle = this.heroSection.getByRole(
            'heading',
            {
                level: 5
            }
        );

        this.heroImage = this.heroSection.locator(
            'picture img'
        );

        // Making Travel Retail Fly section
        this.introductionHeading = page.getByRole(
            'heading',
            {
                name: /making travel retail fly/i
            }
        );

        // Brand logos section
        this.brandLogoSection = page.locator(
            '.paragraph--type--c-teasercycle'
        );

        this.brandLogos = this.brandLogoSection.locator(
            'img.image-style-logo-md'
        );

        // View All Products button
        this.viewAllProductsButton = page.locator(
            'a.btn.btn-primary[href="/products"]'
        );

        // Travel Retail Business Lounge section
        this.businessLoungeHeading = page.getByRole(
            'heading',
            {
                name: /welcome to travel retails business lounge/i
            }
        );

        this.businessLoungeSection =
            this.businessLoungeHeading.locator(
                'xpath=ancestor::div[contains(@class, "paragraph")][1]'
            );

        this.businessLoungeImage =
            this.businessLoungeSection.locator(
                'img.image-style-text-image'
            );

        // Prepare for Take Off button
        this.prepareForTakeOffButton = page.locator(
            'a.btn[href="/user/login"]'
        );
    }

    async navigate(pagePath) {
        await this.page.goto(pagePath, {
            waitUntil: 'domcontentloaded'
        });

        await this.cookieConsent.acceptAllCookies();
    }

    async verifyHomePageNavigation(
        expectedTitle,
        expectedBreadcrumb
    ) {
        await expect(this.page).toHaveURL(/\/$/);

        await expect(this.page).toHaveTitle(
            new RegExp(expectedTitle, 'i')
        );

        await expect(this.breadcrumb).toBeVisible();

        await expect(this.breadcrumb).toContainText(
            expectedBreadcrumb
        );
    }

    async verifyHeroSection(expectedHeadings) {
        await expect(this.heroSection).toBeVisible();

        await expect(this.heroHeading).toHaveText(
            expectedHeadings.hero
        );

        await expect(this.heroSubtitle).toHaveText(
            expectedHeadings.heroSubtitle
        );

        await expect(this.heroImage).toBeVisible();
    }

    async verifyIntroductionSection(expectedHeading) {
        await expect(
            this.introductionHeading
        ).toBeVisible();

        await expect(
            this.introductionHeading
        ).toHaveText(expectedHeading);
    }

    async verifyBrandLogos(minimumLogoCount) {
        await this.brandLogoSection.scrollIntoViewIfNeeded();

        await expect(
            this.brandLogoSection
        ).toBeVisible();

        const totalLogos = await this.brandLogos.count();

        expect(
            totalLogos,
            `Expected at least ${minimumLogoCount} brand logo`
        ).toBeGreaterThanOrEqual(minimumLogoCount);

        for (
            let index = 0;
            index < totalLogos;
            index += 1
        ) {
            await expect(
                this.brandLogos.nth(index)
            ).toBeVisible();
        }
    }

    async verifyViewAllProductsButton() {
        await expect(
            this.viewAllProductsButton
        ).toBeVisible();

        await expect(
            this.viewAllProductsButton
        ).toBeEnabled();

        await expect(
            this.viewAllProductsButton
        ).toHaveAttribute('href', '/products');
    }

    async navigateToProducts() {
        await this.cookieConsent.acceptAllCookies();

        await this.viewAllProductsButton.click();
    }

    async verifyBusinessLoungeSection(
        expectedHeading
    ) {
        await this.businessLoungeHeading
            .scrollIntoViewIfNeeded();

        await expect(
            this.businessLoungeHeading
        ).toBeVisible();

        await expect(
            this.businessLoungeHeading
        ).toHaveText(expectedHeading);

        await expect(
            this.businessLoungeImage
        ).toBeVisible();

        await expect(
            this.prepareForTakeOffButton
        ).toBeVisible();

        await expect(
            this.prepareForTakeOffButton
        ).toHaveAttribute(
            'href',
            '/user/login'
        );
    }

    async verifyPrepareForTakeOffButton() {
        await this.prepareForTakeOffButton
            .scrollIntoViewIfNeeded();

        await expect(
            this.prepareForTakeOffButton
        ).toBeVisible();

        await expect(
            this.prepareForTakeOffButton
        ).toBeEnabled();

        await expect(
            this.prepareForTakeOffButton
        ).toHaveAttribute(
            'href',
            '/user/login'
        );
    }

    async navigateToUserLogin() {
        await this.cookieConsent.acceptAllCookies();

        await this.prepareForTakeOffButton.click();
    }
}

const { expect } = require('@playwright/test');

class B2BLoginPage {
    constructor(page) {
        this.page = page;

        // Public page
        this.siteHeader = page.locator('#header');

        this.b2bLoginLink = this.siteHeader
            .locator('a.nav-link--login-modal')
            .first();

        // Drupal Login modal
        this.loginDialog = page.locator(
            'div.ui-dialog[role="dialog"]'
        );

        this.modalContent = this.loginDialog.locator(
            '#drupal-modal'
        );

        this.loginForm = this.modalContent.locator(
            'form#custom-popup-login-form'
        );

        // Login form controls
        this.usernameInput = this.loginForm.locator(
            'input[name="name"], input[type="text"]'
        ).first();

        this.passwordInput = this.loginForm.locator(
            'input[name="pass"], input[type="password"]'
        ).first();

        this.loginButton = this.loginForm.getByRole(
            'button',
            {
                name: 'Login',
                exact: true
            }
        );

        this.forgotPasswordLink = this.loginForm.getByRole(
            'link',
            {
                name: /forgot password/i
            }
        );

        this.modalCloseButton = this.loginDialog.locator(
            '.ui-dialog-titlebar-close'
        );

        // OneTrust Cookie Banner
        this.oneTrustContainer = page.locator(
            '#onetrust-consent-sdk'
        );

        this.cookieBanner = page.locator(
            '#onetrust-banner-sdk'
        );

        this.acceptAllCookiesButton = page.locator(
            '#onetrust-accept-btn-handler'
        );

        this.rejectAllCookiesButton = page.locator(
            '#onetrust-reject-all-handler'
        );

        // OneTrust Preference Center
        this.preferenceCenter = page.locator(
            '#onetrust-pc-sdk'
        );

        this.acceptRecommendedButton = page.locator(
            '#accept-recommended-btn-handler'
        );

        this.savePreferencesButton = page.locator(
            '.save-preference-btn-handler'
        );

        this.closePreferencesButton = page.locator(
            '#close-pc-btn-handler'
        );

        this.oneTrustDarkOverlay = page.locator(
            '.onetrust-pc-dark-filter'
        );

        // Authenticated Dashboard
        this.dashboardContent = page.locator(
            '#block-retailx-dashboardmenu'
        );
    }

    async isVisible(locator, timeout = 1500) {
        return locator.isVisible({
            timeout
        }).catch(() => false);
    }

    async clickIfVisible(locator) {
        const visible = await this.isVisible(
            locator
        );

        if (!visible) {
            return false;
        }

        await locator.click({
            force: true,
            timeout: 5000
        }).catch(() => {});

        return true;
    }

    async closeOneTrustPreferenceCenter() {
        const preferenceCenterVisible =
            await this.isVisible(
                this.preferenceCenter
            );

        const overlayVisible =
            await this.isVisible(
                this.oneTrustDarkOverlay
            );

        if (
            !preferenceCenterVisible &&
            !overlayVisible
        ) {
            return;
        }

        const acceptedRecommended =
            await this.clickIfVisible(
                this.acceptRecommendedButton
            );

        if (!acceptedRecommended) {
            const savedPreferences =
                await this.clickIfVisible(
                    this.savePreferencesButton
                );

            if (!savedPreferences) {
                await this.clickIfVisible(
                    this.closePreferencesButton
                );
            }
        }

        await this.preferenceCenter.waitFor({
            state: 'hidden',
            timeout: 5000
        }).catch(() => {});
    }

    async closeOneTrustCookieBanner() {
        const bannerVisible =
            await this.isVisible(
                this.cookieBanner
            );

        if (!bannerVisible) {
            return;
        }

        const accepted =
            await this.clickIfVisible(
                this.acceptAllCookiesButton
            );

        if (!accepted) {
            await this.clickIfVisible(
                this.rejectAllCookiesButton
            );
        }

        await this.cookieBanner.waitFor({
            state: 'hidden',
            timeout: 5000
        }).catch(() => {});
    }

    async ensureOneTrustIsClosed() {
        await this.closeOneTrustPreferenceCenter();

        await this.closeOneTrustCookieBanner();

        // Recheck because OneTrust can reopen the Preference Center.
        await this.closeOneTrustPreferenceCenter();

        const overlayVisible =
            await this.isVisible(
                this.oneTrustDarkOverlay
            );

        if (overlayVisible) {
            await this.clickIfVisible(
                this.closePreferencesButton
            );

            await this.clickIfVisible(
                this.savePreferencesButton
            );

            await this.clickIfVisible(
                this.acceptRecommendedButton
            );
        }

        await this.oneTrustDarkOverlay.waitFor({
            state: 'hidden',
            timeout: 5000
        }).catch(() => {});
    }

    async navigate(pagePath = '/') {
        await this.page.goto(pagePath, {
            waitUntil: 'domcontentloaded',
            timeout: 60000
        });

        await expect(
            this.siteHeader,
            'Site Header should be displayed'
        ).toBeVisible({
            timeout: 30000
        });

        await this.ensureOneTrustIsClosed();
    }

    async openLoginModal() {
        await this.ensureOneTrustIsClosed();

        await expect(
            this.b2bLoginLink,
            'B2B Log in link should be visible'
        ).toBeVisible({
            timeout: 15000
        });

        await expect(
            this.b2bLoginLink,
            'B2B Log in link should be enabled'
        ).toBeEnabled();

        const overlayBeforeModal =
            await this.isVisible(
                this.oneTrustDarkOverlay
            );

        if (overlayBeforeModal) {
            await this.ensureOneTrustIsClosed();
        }

        await this.b2bLoginLink.click({
            timeout: 15000,
            force: true
        });

        await expect(
            this.loginDialog,
            'B2B Login dialog should open'
        ).toBeVisible({
            timeout: 15000
        });

        await expect(
            this.loginForm,
            'B2B Login form should be displayed'
        ).toBeVisible({
            timeout: 15000
        });

        // OneTrust can reappear after Drupal opens the modal.
        await this.ensureOneTrustIsClosed();
    }

    async verifyLoginModal() {
        await expect(
            this.loginDialog
        ).toBeVisible();

        await expect(
            this.loginForm
        ).toBeVisible();

        await expect(
            this.usernameInput
        ).toBeVisible();

        await expect(
            this.passwordInput
        ).toBeVisible();

        await expect(
            this.loginButton
        ).toBeVisible();

        await expect(
            this.modalCloseButton
        ).toBeVisible();
    }

    async verifyFieldsInitiallyEmpty() {
        await expect(
            this.usernameInput
        ).toHaveValue('');

        await expect(
            this.passwordInput
        ).toHaveValue('');
    }

    async verifyForgotPasswordLink() {
        await expect(
            this.forgotPasswordLink
        ).toBeVisible();

        await expect(
            this.forgotPasswordLink
        ).toHaveAttribute(
            'href',
            /password|forgot/i
        );
    }

    async fillCredentials(
        username,
        password
    ) {
        expect(
            username,
            'B2B username should be provided'
        ).toBeTruthy();

        expect(
            password,
            'B2B password should be provided'
        ).toBeTruthy();

        await expect(
            this.usernameInput
        ).toBeVisible();

        await expect(
            this.passwordInput
        ).toBeVisible();

        await this.usernameInput.fill(
            username
        );

        await this.passwordInput.fill(
            password
        );
    }

    async verifyCredentialsEntered() {
        const usernameValue =
            await this.usernameInput.inputValue();

        const passwordValue =
            await this.passwordInput.inputValue();

        expect(
            usernameValue.length,
            'Username field should contain a value'
        ).toBeGreaterThan(0);

        expect(
            passwordValue.length,
            'Password field should contain a value'
        ).toBeGreaterThan(0);

        await expect(
            this.passwordInput
        ).toHaveAttribute(
            'type',
            'password'
        );
    }

    async submitLogin(
        expectedDashboardPath = '/dashboard'
    ) {
        // OneTrust can reappear after credentials are entered.
        await this.ensureOneTrustIsClosed();

        await expect(
            this.loginButton,
            'B2B Login button should be visible'
        ).toBeVisible({
            timeout: 15000
        });

        await expect(
            this.loginButton,
            'B2B Login button should be enabled'
        ).toBeEnabled();

        const overlayBeforeSubmit =
            await this.isVisible(
                this.oneTrustDarkOverlay
            );

        if (overlayBeforeSubmit) {
            await this.ensureOneTrustIsClosed();
        }

        const navigationPromise =
            this.page.waitForURL(
                url =>
                    url.pathname ===
                    expectedDashboardPath,
                {
                    timeout: 30000
                }
            );

        /*
         * force:true is intentional here.
         * The button is already verified as visible and enabled.
         * This prevents a delayed OneTrust backdrop from intercepting
         * the Login click after the final overlay check.
         */
        await this.loginButton.click({
            timeout: 15000,
            force: true
        });

        await navigationPromise;

        await expect(this.page).toHaveURL(
            url =>
                url.pathname ===
                expectedDashboardPath
        );

        await expect(
            this.loginDialog
        ).toBeHidden({
            timeout: 15000
        });

        await expect(
            this.dashboardContent,
            'Authenticated Dashboard navigation should be displayed'
        ).toBeVisible({
            timeout: 30000
        });
    }

    async closeLoginModal() {
        await expect(
            this.modalCloseButton
        ).toBeVisible();

        await this.modalCloseButton.click({
            force: true
        });

        await expect(
            this.loginDialog
        ).toBeHidden({
            timeout: 15000
        });
    }
}
